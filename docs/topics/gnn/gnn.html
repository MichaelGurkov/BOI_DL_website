# === Utilities (shared helpers) ===

def seed_everything(seed: int = 7) -> None:
    """
    Best-effort seeding across numpy/torch for reproducibility.
    """
    import os
    import random

    os.environ["PYTHONHASHSEED"] = str(seed)
    random.seed(seed)
    np.random.seed(seed)

    torch.manual_seed(seed)
    if torch.cuda.is_available():
        torch.cuda.manual_seed_all(seed)

    # Determinism (can slow down execution)
    torch.backends.cudnn.deterministic = True
    torch.backends.cudnn.benchmark = False


def get_device() -> str:
    """
    Returns "cuda" if available, else "cpu".
    """
    return "cuda" if torch.cuda.is_available() else "cpu"


def make_pos_weight(y_true: torch.Tensor) -> torch.Tensor:
    """
    Compute pos_weight for BCEWithLogitsLoss for binary labels in {0,1}.
    pos_weight = (#neg / #pos), returned as a tensor of shape [1].
    """
    y_true = y_true.float()
    pos = (y_true == 1).sum().item()
    neg = (y_true == 0).sum().item()
    return torch.tensor([neg / max(pos, 1.0)], device=y_true.device)


def eval_probs(probs: torch.Tensor, y_true: torch.Tensor) -> tuple[float, float]:
    """
    Compute ROC-AUC and PR-AUC from predicted probabilities and true labels.
    """
    p = probs.detach().cpu().numpy()
    yt = y_true.detach().cpu().numpy()
    return roc_auc_score(yt, p), average_precision_score(yt, p)


def print_split_summary(
    time_step: np.ndarray,
    train_mask: np.ndarray,
    val_mask: np.ndarray,
    test_mask: np.ndarray
) -> None:
    """
    Print split sizes and time-step ranges for sanity checking.
    """
    print(
        "Split sizes (labeled):",
        "train", int(train_mask.sum()),
        "val", int(val_mask.sum()),
        "test", int(test_mask.sum())
    )

    print("time_step range overall:",
          int(time_step.min()), int(time_step.max()))
    print("train time_step min/max:",
          int(time_step[train_mask].min()), int(time_step[train_mask].max()))
    print("val   time_step min/max:",
          int(time_step[val_mask].min()), int(time_step[val_mask].max()))
    print("test  time_step min/max:",
          int(time_step[test_mask].min()), int(time_step[test_mask].max()))

def train_binary_classifier(
    model: nn.Module,
    forward_fn,
    y: torch.Tensor,
    train_mask: torch.Tensor,
    val_mask: torch.Tensor,
    test_mask: torch.Tensor,
    epochs: int = 30,
    lr: float = 1e-3,
    weight_decay: float = 1e-4,
    log_every: int = 5,
):
    """
    Generic trainer for binary classification with BCEWithLogitsLoss + pos_weight.
    forward_fn(model) must return logits for ALL nodes/samples (shape [N]).
    """

    # pos_weight computed on training labels only
    ytr = y[train_mask].float()
    pos_weight = make_pos_weight(ytr)

    crit = nn.BCEWithLogitsLoss(pos_weight=pos_weight)
    opt = optim.Adam(model.parameters(), lr=lr, weight_decay=weight_decay)

    best_pr, best_state = -1.0, None

    for ep in range(1, epochs + 1):
        model.train()
        opt.zero_grad()

        logits = forward_fn(model)
        loss = crit(logits[train_mask], y[train_mask].float())

        loss.backward()
        opt.step()

        if ep == 1 or ep % log_every == 0:
            model.eval()
            with torch.no_grad():
                p_val = torch.sigmoid(forward_fn(model)[val_mask])
            val_roc, val_pr = eval_probs(p_val, y[val_mask])

            print(
                f"Epoch {ep:03d} | loss {loss.item():.4f} | "
                f"val ROC-AUC {val_roc:.4f} | val PR-AUC {val_pr:.4f}"
            )

            if val_pr > best_pr:
                best_pr = val_pr
                best_state = {k: v.detach().cpu().clone() for k, v in model.state_dict().items()}

    if best_state is not None:
        model.load_state_dict(best_state)

    model.eval()
    with torch.no_grad():
        p_test = torch.sigmoid(forward_fn(model)[test_mask])
    test_roc, test_pr = eval_probs(p_test, y[test_mask])

    return best_pr, test_roc, test_pr

# === Models (definitions only) ===

class MLP(nn.Module):
    def __init__(self, in_dim: int, hidden: int = 128, drop: float = 0.3):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(in_dim, hidden),
            nn.ReLU(),
            nn.Dropout(drop),
            nn.Linear(hidden, hidden),
            nn.ReLU(),
            nn.Dropout(drop),
            nn.Linear(hidden, 1)
        )

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return self.net(x).squeeze(-1)


class GraphSAGE(nn.Module):
    def __init__(self, in_dim: int, hidden: int = 64, drop: float = 0.3):
        super().__init__()
        self.conv1 = SAGEConv(in_dim, hidden)
        self.conv2 = SAGEConv(hidden, 1)
        self.drop = drop

    def forward(self, x: torch.Tensor, edge_index: torch.Tensor) -> torch.Tensor:
        x = self.conv1(x, edge_index)
        x = F.relu(x)
        x = F.dropout(x, p=self.drop, training=self.training)
        x = self.conv2(x, edge_index)
        return x.squeeze(-1)

Overview

This notebook uses the Elliptic Bitcoin Dataset, a widely used benchmark for anti–money laundering (AML) research on blockchain transaction networks.
The dataset was constructed to support graph-based, semi-supervised learning, and is commonly used to evaluate Graph Neural Networks (GNNs) and other relational models.

Graph Structure

The data is organized as a directed transaction graph:

  • Nodes represent individual Bitcoin transactions (not users or wallets).
  • Directed edges represent flows of funds between transactions:
    an edge from transaction A to transaction B means that an output of A is used as an input to B.
  • The graph is acyclic and time-ordered, reflecting the causal structure of Bitcoin transactions.

Each transaction belongs to a discrete time step.
Edges always point forward in time, and transactions from future time steps never influence past ones.
This temporal structure is critical to avoid information leakage when training models.

Labels

A subset of transactions is labeled by Elliptic’s analysts:

  • Illicit (label = 1)
    Transactions associated with known illegal activity, such as scams, ransomware, darknet markets, Ponzi schemes, and malware-related operations.
  • Licit (label = 2)
    Transactions associated with legitimate services, such as exchanges, wallet providers, miners, payment services, and other lawful entities.
  • Unknown (unlabeled)
    Transactions for which no reliable classification was available.

Only a minority of nodes are labeled, making the task semi-supervised by nature.
In most experiments, models are trained only on labeled nodes while leveraging the full graph structure.

Feature Representation

Each node is associated with a fixed-length feature vector (166 features in total) that describes properties of the transaction and its local network context.
All features are hand-engineered, anonymized, and numerical.

The features fall into three main groups:

1. Local Transaction Features

These features describe intrinsic properties of the transaction itself, without reference to neighboring transactions.

They include statistics such as:

  • Number of inputs and outputs

  • Transaction value aggregates

  • Fee-related measures

  • Simple structural characteristics of the transaction

These features capture how the transaction is constructed on the Bitcoin protocol level.

2. Aggregated Neighborhood Features

These features summarize information about the immediate transaction neighborhood, computed over transactions connected by incoming and outgoing edges.
They include statistical aggregates (e.g., sums, means, counts, variances) over:

  • Neighbor transaction values

  • Neighbor degrees

  • Flow patterns into and out of the transaction

Importantly, these aggregations are direction-aware (incoming vs. outgoing flows), reflecting how funds enter and leave the transaction.

3. Temporal Aggregation Features

The dataset incorporates time-aware aggregation, where neighborhood statistics are computed over previous time steps only.
This ensures that each transaction’s features are based solely on information that would have been available at the time the transaction occurred.

This design choice is crucial:

  • It preserves causality

  • It prevents models from exploiting future information

  • It makes the dataset suitable for realistic AML scenarios

Learning Task

The standard task is binary node classification: > Predict whether a transaction is licit or illicit, using node features and the transaction graph.

Key characteristics of the task:

  • Severe class imbalance (illicit transactions are rare)

  • Partial labeling (many nodes are unlabeled)

  • Strong relational signal (labels are correlated along transaction flows)

Because of this, performance is typically evaluated using metrics such as ROC-AUC or PR-AUC, rather than raw accuracy.

Overall, the Elliptic dataset provides a realistic and challenging setting for studying graph-based fraud detection, combining rich feature engineering, temporal constraints, and real-world noise.

Why this is a graph learning problem

A transaction’s risk is often not determined only by its own attributes, but also by its neighborhood in the flow-of-funds network:

  • The same transaction features can mean different things depending on who paid whom.

  • Illicit behavior can appear as patterns over multiple hops (e.g., fan-in/fan-out, chains, mixing-like structures).

  • Graph Neural Networks (GNNs) learn from both:

    • Node features (attributes), and

    • Graph structure (who is connected to whom)

In this exercise, the core task is node classification: > Predict whether a transaction node is illicit or licit.

Baseline model: MLP

The baseline model used in this notebook is a standard feed-forward neural network that operates solely on the transaction feature vectors, without using any information about the transaction graph. Each transaction is treated as an independent sample, and its features are passed through several fully connected layers with nonlinear activations to produce a probability estimate of whether the transaction is illicit or licit. This model captures relationships and interactions among the numerical attributes of a transaction, but it is entirely blind to how transactions are connected to one another through the flow of funds. As such, it represents the strongest reasonable non-graph baseline: a model that can learn complex nonlinear decision boundaries from features alone, yet cannot exploit network structure. Comparing graph-based models against this baseline makes it possible to isolate the added value of graph information. If a graph neural network outperforms this model, the improvement can be attributed to learning from the connectivity and neighborhood patterns rather than from features alone.

SAGE

The graph-based model used in this notebook is a neighborhood-aggregation neural network designed to explicitly exploit the transaction graph structure. Instead of treating each transaction in isolation, the model updates a transaction’s representation by combining its own feature vector with information propagated from directly connected transactions in the graph. At each layer, a transaction aggregates representations from its neighbors along the flow-of-funds edges and transforms the result through a learnable function, allowing information to travel across multiple hops as layers are stacked. This mechanism enables the model to capture relational patterns such as chains, fan-in/fan-out structures, and clusters of related activity that are characteristic of illicit behavior but are invisible to feature-only models. By learning jointly from node attributes and graph connectivity, the model can assign risk scores that reflect both local transaction properties and their broader network context. In this notebook, this approach is implemented using the GraphSAGE architecture, which serves as a representative and scalable example of a Graph Neural Network for node classification in transaction graphs.

k_hops   = 6
nodes_k = {seed}
frontier = {seed}

for _ in range(k_hops):

    for n in frontier:

        nxt |= set(G.successors(n))

    frontier   = nxt
